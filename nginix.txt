server {
    listen 80;
    server_name claudify.com www.claudify.com;

    # 1. A Vite által lefordított (dist) frontend mappa helye
    root /var/www/claudify/dist;
    index index.html;

    # A React (kliensoldali) útválasztás biztosítása
    location / {
        try_files $uri $uri/ /index.html;
    }

    # 2. API kérések átirányítása a Node.js backendre
    location /api/ {
        proxy_pass http://127.0.0.1:3000/; # A Node.js szervered portja
        proxy_http_version 1.1;
        
        # Websocket és folyamatos kapcsolatok támogatása
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection 'upgrade';
        
        # Eredeti kliens adatok továbbítása
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;

        # AI streaming (SSE) miatt kritikus beállítás: kikapcsolja a pufferelést
        proxy_buffering off;
        proxy_read_timeout 300s;
    }
}
